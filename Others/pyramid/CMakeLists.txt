CMAKE_MINIMUM_REQUIRED(VERSION 3.15)

PROJECT(samples)

SET(CMAKE_CXX_STANDARD 17)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

FIND_PACKAGE(OpenSceneGraph REQUIRED osgViewer)

INCLUDE_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR} ${OPENSCENEGRAPH_INCLUDE_DIRS})
LINK_DIRECTORIES(${CMAKE_CURRENT_SOURCE_DIR})
ADD_EXECUTABLE(pyramid pyramid.cc graphics.cc)
TARGET_LINK_LIBRARIES(pyramid ${OPENSCENEGRAPH_LIBRARIES})
